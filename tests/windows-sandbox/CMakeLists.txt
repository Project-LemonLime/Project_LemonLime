add_executable(sandbox-helper helper.cpp)
add_executable(sandbox-static-memory-helper helper.cpp)
target_compile_definitions(sandbox-static-memory-helper PRIVATE LEMON_TEST_STATIC_MEMORY)
foreach(helper IN ITEMS sandbox-helper sandbox-static-memory-helper)
    target_compile_definitions(${helper} PRIVATE WIN32_LEAN_AND_MEAN)
    set_target_properties(${helper} PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    if(MSVC)
        set_property(TARGET ${helper} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    if(MINGW)
        target_link_options(${helper} PRIVATE -municode -static)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_compile_options(${helper} PRIVATE -fms-extensions)
        endif()
    endif()
    target_link_libraries(${helper} PRIVATE advapi32 ws2_32 psapi)
endforeach()

add_executable(windows-sandbox-tests main.cpp)
target_compile_definitions(windows-sandbox-tests PRIVATE WIN32_LEAN_AND_MEAN)
target_include_directories(windows-sandbox-tests PRIVATE ${CMAKE_SOURCE_DIR}/src ${spdlog_DIR}/include)
target_link_libraries(windows-sandbox-tests PRIVATE lemon-core SingleApplication::SingleApplication ${LEMON_QT_LIBNAME}::Test spdlog::spdlog ws2_32)
add_dependencies(windows-sandbox-tests sandbox-helper sandbox-static-memory-helper)
add_test(NAME windows-sandbox COMMAND windows-sandbox-tests -o sandbox-results.txt,txt -o -,txt)
set_tests_properties(windows-sandbox PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} TIMEOUT 180)
